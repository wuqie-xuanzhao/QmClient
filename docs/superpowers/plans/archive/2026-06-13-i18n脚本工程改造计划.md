**Goal:** 把 `qmclient_scripts/languages_qmclient/` 从迁移期脚本集合改造成一条可信的“源码 key 提取 → 语言文件同步 → 审查报告 → 校验门禁”流水线。

**Tech Stack:** Python 3.14、`tomllib`、DDNet language txt 格式、`unittest`、`ruff`、`qmclient_scripts/gate/check_gate.py`。

## Design Rule: 简中翻译维护库不是 txt 镜像

`qmclient_scripts/languages_qmclient/translations/simplified_chinese.toml` 是 QmClient 的翻译维护层，不是 `data/languages/simplified_chinese.txt` 换格式后的全量副本。

维护库应该收：

- QmClient 自己新增的 key。
- 从 `data/qmclient/languages/` overlay 迁移来的 key。
- QmClient 想覆盖或修正上游译文的 key。
- 当前暂时未使用但未来可能复用的 QmClient 专属译文。
- 需要备注翻译理由的 key，例如服务器消息、Dream 梗、UI 语气。

维护库不应该收：

- DDNet 上游已有稳定中文翻译、QmClient 不需要改的 key。
- 纯 passthrough 的品牌名、API 名、配置字段名。
- 全 `src` 提取出来但只是普通上游 UI 的大批文案。

运行时 `data/languages/simplified_chinese.txt` 是生成产物：它可以包含所有当前源码需要的 key，但不承载“长期保存不用译文”的职责。

Create `qmclient_scripts/languages_qmclient/import_simplified_chinese_translations.py`:

```python
#!/usr/bin/env python3
"""Import Simplified Chinese translations into the QmClient-maintained TOML library."""

from __future__ import annotations

import subprocess
from collections import OrderedDict
from pathlib import Path

SCRIPT_DIR = Path(__file__).resolve().parent
PROJECT_ROOT = SCRIPT_DIR.parents[1]
CURRENT_SIMPLIFIED = PROJECT_ROOT / "data" / "languages" / "simplified_chinese.txt"
OUTPUT = SCRIPT_DIR / "translations" / "simplified_chinese.toml"

HISTORICAL_SOURCES = {
    "legacy-overlay": "0b2b40da2:data/qmclient/languages/simplified_chinese.txt",
    "upstream-base": "a4a8eb09a:data/languages/simplified_chinese.txt",
}


def read_language_pairs(path: Path) -> dict[str, str]:
    pairs: dict[str, str] = {}
    current_key: str | None = None
    with path.open(encoding="utf-8-sig") as file:
        for raw_line in file:
            line = raw_line.rstrip("\r\n")
            if not line or line.startswith("#"):
                current_key = None
                continue
            if line.startswith("[") and not line.startswith("[%"):
                current_key = None
                continue
            if line.startswith("== "):
                if current_key is not None:
                    pairs[current_key] = line[3:]
                current_key = None
                continue
            current_key = line
    return pairs


def read_git_language_pairs(revision_path: str) -> dict[str, str]:
    result = subprocess.run(
        ["git", "show", revision_path],
        cwd=PROJECT_ROOT,
        check=True,
        text=True,
        encoding="utf-8",
        stdout=subprocess.PIPE,
    )
    tmp = SCRIPT_DIR / ".simplified_chinese_import_tmp.txt"
    try:
        tmp.write_text(result.stdout, encoding="utf-8")
        return read_language_pairs(tmp)
    finally:
        if tmp.exists():
            tmp.unlink()


def should_keep_translation(key: str, translation: str, source_name: str) -> bool:
    if not translation or translation == key:
        return False
    if source_name == "current" and key == translation:
        return False
    return source_name in {"legacy-overlay", "qmclient", "current-override"}


def merge_candidates(sources: list[tuple[str, dict[str, str]]]) -> OrderedDict[str, dict[str, str]]:
    merged: OrderedDict[str, dict[str, str]] = OrderedDict()
    for source_name, pairs in sources:
        for key, translation in pairs.items():
            if not should_keep_translation(key, translation, source_name):
                continue
            merged[key] = {
                "translation": translation,
                "status": "draft" if source_name != "legacy-overlay" else "reviewed",
                "source": source_name,
                "note": "",
            }
    return OrderedDict(sorted(merged.items(), key=lambda item: item[0].lower()))


def toml_quote(value: str) -> str:
    return '"' + value.replace("\\", "\\\\").replace('"', '\\"') + '"'


def dump_translation_toml(entries: dict[str, dict[str, str]]) -> str:
    lines: list[str] = [
        "# QmClient Simplified Chinese translation maintenance library.",
        "# This is not a full mirror of data/languages/simplified_chinese.txt.",
        "# Keep QmClient-specific, overlay-migrated, or intentional override translations here.",
        "",
    ]
    for key, entry in entries.items():
        lines.append(f"[{toml_quote(key)}]")
        lines.append(f"translation = {toml_quote(entry['translation'])}")
        lines.append(f"status = {toml_quote(entry['status'])}")
        lines.append(f"source = {toml_quote(entry['source'])}")
        lines.append(f"note = {toml_quote(entry.get('note', ''))}")
        lines.append("")
    return "\n".join(lines)


def import_translations() -> OrderedDict[str, dict[str, str]]:
    sources: list[tuple[str, dict[str, str]]] = []
    for source_name, revision_path in HISTORICAL_SOURCES.items():
        sources.append((source_name, read_git_language_pairs(revision_path)))
    sources.append(("current", read_language_pairs(CURRENT_SIMPLIFIED)))
    return merge_candidates(sources)


def main() -> None:
    entries = import_translations()
    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(dump_translation_toml(entries), encoding="utf-8", newline="\n")
    print(f"Wrote {len(entries)} maintained translations to {OUTPUT.relative_to(PROJECT_ROOT)}")


if __name__ == "__main__":
    main()
```

```python
    "%c Team %d",
    "%d players",
    "%d teams",

def iter_source_files(paths: tuple[Path, ...] = SOURCE_PATHS) -> list[Path]:
    files: list[Path] = []
    for root in paths:
        if root.is_file():
            files.append(root)
            continue
        for path in sorted(root.rglob("*")):
            if path.suffix in SOURCE_EXTENSIONS:
                files.append(path)
    return sorted(files)


def collect_source_keys() -> list[str]:
    keys: set[str] = set(EXTRA_LOCALIZE_STRINGS)
    for path in iter_source_files():
        content = strip_cpp_comments(path.read_text(encoding="utf-8"))
        keys.update(key for key, _context in extract_localize_keys(content))
        keys.update(extract_register_help_strings(content))
        keys.update(extract_known_indirect_strings(path, content))
    return sorted(keys)
```

```python
#!/usr/bin/env python3
"""Extract localization source keys for reporting/generation only."""

from __future__ import annotations

import os
from pathlib import Path

import source_keys

SCRIPT_DIR = Path(__file__).resolve().parent
PROJECT_ROOT = SCRIPT_DIR.parents[1]
STRINGS_FILE = SCRIPT_DIR / "extracted_strings.txt"


def collect_strings() -> list[str]:
    return source_keys.collect_source_keys()


def classify_string(text: str) -> str:
    if any("\u3400" <= ch <= "\u4dbf" or "\u4e00" <= ch <= "\u9fff" for ch in text):
        return "needs_review"
    return "key_or_non_zh"


def main() -> None:
    sorted_strings = collect_strings()
    with STRINGS_FILE.open("w", encoding="utf-8", newline="\n") as file:
        for key in sorted_strings:
            file.write(key + "\n")

    rel_outpath = os.path.relpath(STRINGS_FILE, PROJECT_ROOT)
    print(f"Extracted {len(sorted_strings)} unique localization strings to {rel_outpath}")
    for index, key in enumerate(sorted_strings, start=1):
        print(f"  {index:3d}. {key}")


if __name__ == "__main__":
    main()
```
